# Aila v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a minimal Expo iOS app that authenticates a Danish guardian via MitID and displays their Aula profile (name, institution, children), proving the API connection works.

**Architecture:** Single Expo app with Expo Router for navigation. No backend. Auth via `expo-web-browser.openAuthSessionAsync` with PKCE against Aula's OAuth endpoints. Tokens stored in iOS Keychain via `expo-secure-store`. Direct HTTP calls to `https://app-private.aula.dk/api/v22`.

**Tech Stack:** Expo SDK 52, Expo Router, TypeScript, expo-web-browser, expo-secure-store, expo-crypto, react-native-reanimated, react-native-gesture-handler, Jest + jest-expo

// lib/token-storage.ts
import * as SecureStore from 'expo-secure-store';

const KEYS = {
  accessToken: 'aula_access_token',
  refreshToken: 'aula_refresh_token',
} as const;

export interface StoredTokens {
  accessToken: string;
  refreshToken: string;
}

export async function storeTokens(tokens: StoredTokens): Promise<void> {
  await SecureStore.setItemAsync(KEYS.accessToken, tokens.accessToken);
  await SecureStore.setItemAsync(KEYS.refreshToken, tokens.refreshToken);
}

export async function getStoredTokens(): Promise<StoredTokens | null> {
  const accessToken = await SecureStore.getItemAsync(KEYS.accessToken);
  const refreshToken = await SecureStore.getItemAsync(KEYS.refreshToken);
  if (!accessToken || !refreshToken) return null;
  return { accessToken, refreshToken };
}

export async function clearTokens(): Promise<void> {
  await SecureStore.deleteItemAsync(KEYS.accessToken);
  await SecureStore.deleteItemAsync(KEYS.refreshToken);
}
```

// lib/pkce.ts
import * as Crypto from 'expo-crypto';

export function generateCodeVerifier(): string {
  const array = new Uint8Array(32);
  Crypto.getRandomValues(array);
  return btoa(String.fromCharCode(...Array.from(array)))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=/g, '');
}

export async function generateCodeChallenge(verifier: string): Promise<string> {
  const digest = await Crypto.digestStringAsync(
    Crypto.CryptoDigestAlgorithm.SHA256,
    verifier,
    { encoding: Crypto.CryptoEncoding.BASE64 },
  );
  return digest
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=/g, '');
}
```

// lib/auth.ts
import * as WebBrowser from 'expo-web-browser';
import { AULA_OAUTH } from '../constants/aula';
import { generateCodeVerifier, generateCodeChallenge } from './pkce';
import {
  storeTokens,
  getStoredTokens,
  clearTokens,
  StoredTokens,
} from './token-storage';

interface TokenResponse {
  access_token: string;
  refresh_token: string;
}

export async function startMitIdLogin(): Promise<StoredTokens> {
  const codeVerifier = generateCodeVerifier();
  const codeChallenge = await generateCodeChallenge(codeVerifier);

  const params = new URLSearchParams({
    response_type: 'code',
    client_id: AULA_OAUTH.clientId,
    scope: AULA_OAUTH.scope,
    redirect_uri: AULA_OAUTH.redirectUri,
    code_challenge: codeChallenge,
    code_challenge_method: 'S256',
    state: generateCodeVerifier(),
  });

  const authUrl = `${AULA_OAUTH.authorizationUrl}?${params}`;
  const result = await WebBrowser.openAuthSessionAsync(
    authUrl,
    AULA_OAUTH.redirectUri,
  );

  if (result.type !== 'success') {
    throw new Error('Login annulleret');
  }

  const url = new URL(result.url);
  const code = url.searchParams.get('code');
  if (!code) throw new Error('Intet autoriseringskode i svar');

  return exchangeCodeForTokens(code, codeVerifier);
}

export async function exchangeCodeForTokens(
  code: string,
  codeVerifier: string,
): Promise<StoredTokens> {
  const body = new URLSearchParams({
    grant_type: 'authorization_code',
    code,
    client_id: AULA_OAUTH.clientId,
    redirect_uri: AULA_OAUTH.redirectUri,
    code_verifier: codeVerifier,
  });

  const response = await fetch(AULA_OAUTH.tokenUrl, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body.toString(),
  });

  if (!response.ok) {
    throw new Error(`Token-udveksling fejlede: ${response.status}`);
  }

  const data: TokenResponse = await response.json();
  const tokens: StoredTokens = {
    accessToken: data.access_token,
    refreshToken: data.refresh_token,
  };
  await storeTokens(tokens);
  return tokens;
}

export async function refreshTokens(): Promise<StoredTokens> {
  const stored = await getStoredTokens();
  if (!stored) throw new Error('Ingen gemte tokens');

  const body = new URLSearchParams({
    grant_type: 'refresh_token',
    refresh_token: stored.refreshToken,
    client_id: AULA_OAUTH.clientId,
  });

  const response = await fetch(AULA_OAUTH.tokenUrl, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body.toString(),
  });

  if (!response.ok) {
    await clearTokens();
    throw new Error('Session udløbet — log ind igen');
  }

  const data: TokenResponse = await response.json();
  const tokens: StoredTokens = {
    accessToken: data.access_token,
    refreshToken: data.refresh_token,
  };
  await storeTokens(tokens);
  return tokens;
}
```

// lib/aula-api.ts
import { AULA_API_BASE, AULA_METHODS } from '../constants/aula';
import { getStoredTokens } from './token-storage';
import { refreshTokens } from './auth';
import type {
  ProfilesByLoginResponse,
  ProfileContextResponse,
} from './aula-types';

async function aulaFetch<T>(
  method: string,
  params: Record<string, string> = {},
  accessToken?: string,
  isRetry = false,
): Promise<T> {
  const token = accessToken ?? (await getStoredTokens())?.accessToken;
  if (!token) throw new Error('Ikke logget ind');

  const query = new URLSearchParams({ method, access_token: token, ...params });
  const response = await fetch(`${AULA_API_BASE}?${query}`);

  if (response.status === 401 && !isRetry) {
    const refreshed = await refreshTokens();
    return aulaFetch<T>(method, params, refreshed.accessToken, true);
  }

  if (!response.ok) throw new Error(`Aula API fejl: ${response.status}`);
  return response.json() as Promise<T>;
}

export function getProfilesByLogin(): Promise<ProfilesByLoginResponse> {
  return aulaFetch<ProfilesByLoginResponse>(AULA_METHODS.profilesByLogin);
}

export function getProfileContext(): Promise<ProfileContextResponse> {
  return aulaFetch<ProfileContextResponse>(AULA_METHODS.profileContext, {
    portalrole: 'guardian',
  });
}
```

## Troubleshooting

**"Token exchange failed: 400"** — The `code_verifier`/`code_challenge` pairing is wrong, or the code was already used. Try logging in again fresh.

**"Aula API fejl: 404"** — The API endpoint URL is wrong. Verify `AULA_API_BASE` in `constants/aula.ts`. The current known value is `https://app-private.aula.dk/api/v22`.

**"Login annulleret"** — User cancelled the browser. No action needed, tap the button again.

**Home screen shows wrong data shape** — The Aula API response may differ slightly from the types in `aula-types.ts`. Add `console.log(profileRes)` temporarily in `home.tsx` to inspect the real shape, then update the interfaces.

**Xcode build fails** — Run `npx expo prebuild --clean` then `npx expo run:ios` again.
